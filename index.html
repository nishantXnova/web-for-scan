FastQrAnalyzer.kt — zero-copy ImageAnalysis.Analyzer 
// keep only the freshest frame — never queue
val analysis = ImageAnalysis.Builder()
  .setBackpressureStrategy(ImageAnalysis.STRATEGY_KEEP_ONLY_LATEST)
  .setOutputImageFormat(ImageAnalysis.OUTPUT_IMAGE_FORMAT_YUV_420_888)
  .setTargetResolution(Size(960, 540))
  .build()

// restrict to QR only — skip EAN / Code128 / Aztec / DataMatrix trees
val options = BarcodeScannerOptions.Builder()
  .setBarcodeFormats(Barcode.FORMAT_QR_CODE)
  .build()

val executor = Executors.newSingleThreadExecutor { r ->
  Thread(r, "FastScanAnalyzerThread").apply { priority = Thread.MAX_PRIORITY }
}
